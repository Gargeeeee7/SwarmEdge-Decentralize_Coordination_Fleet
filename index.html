# SwarmEdge: Decentralized AMR Coordination Node
# Runs locally on Raspberry Pi 4/5 or NVIDIA Jetson Nano onboard each robot.

BROADCAST_PORT = 5005
BUFFER_SIZE = 1024
NEIGHBORHOOD_RADIUS = 4.0 # Manhattan distance

class SwarmEdgeNode:
    def __init__(self, robot_id, start_pos, rank):
        self.robot_id = robot_id
        self.pos = list(start_pos)
        self.goal = list(start_pos)
        self.rank = rank
        self.battery = 100.0
        self.age = 0
        self.tick = 0
        self.state = "IDLE"
        self.active_job = None
        self.inbox = []
        self.reserved_cells = set()
        
        # P2P UDP Broadcast Socket
        self.sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        self.sock.setsockopt(socket.SOL_SOCKET, socket.SO_BROADCAST, 1)
        self.sock.bind(("", BROADCAST_PORT))
        self.sock.setblocking(False)

    def calculate_priority(self):
        # Priority increases with waiting age to break deadlocks
        return self.age * 10 + (self.rank + self.tick) % 6

    def broadcast_intent(self, next_cell):
        payload = {
            "type": "INTENT",
            "from": self.robot_id,
            "rank": self.rank,
            "pos": self.pos,
            "next": next_cell,
            "priority": self.calculate_priority(),
            "battery": round(self.battery, 1),
            "tick": self.tick,
            "ttl": 2
        }
        msg = json.dumps(payload).encode('utf-8')
        self.sock.sendto(msg, ('<broadcast>', BROADCAST_PORT))

    def broadcast_bid(self, task_id, cost):
        payload = {
            "type": "BID",
            "from": self.robot_id,
            "task_id": task_id,
            "cost": round(cost, 2),
            "battery": round(self.battery, 1)
        }
        msg = json.dumps(payload).encode('utf-8')
        self.sock.sendto(msg, ('<broadcast>', BROADCAST_PORT))

    def step(self):
        self.tick += 1
        # 1. Drain incoming UDP messages from local P2P mesh
        while True:
            try:
                data, addr = self.sock.recvfrom(BUFFER_SIZE)
                packet = json.loads(data.decode('utf-8'))
                if packet.get("from") != self.robot_id:
                    self.inbox.append(packet)
            except BlockingIOError:
                break
        
        # 2. Local space-time arbitration & A* movement loop...
        print(f"[{self.robot_id}] Step {self.tick} | State: {self.state} | Pos: {self.pos} | Battery: {self.battery}%")

if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="SwarmEdge AMR Node")
    parser.add_argument("--id", default="R1", help="Robot ID (e.g. R1, R2)")
    parser.add_argument("--x", type=int, default=0, help="Initial X coordinate")
    parser.add_argument("--y", type=int, default=5, help="Initial Y coordinate")
    parser.add_argument("--rank", type=int, default=0, help="Tie-breaker rank")
    args = parser.parse_args()

    node = SwarmEdgeNode(args.id, (args.x, args.y), args.rank)
    print(f"SwarmEdge Node {args.id} online. Broadcasting on port {BROADCAST_PORT}...")
    try:
        while True:
            node.step()
            time.sleep(0.5)
    except KeyboardInterrupt:
        print("Node halted.")

sudo apt-get update
sudo apt-get install -y python3 python3-pip net-tools wireless-tools

# 2. Configure Ad-Hoc / Wi-Fi Mesh interface (No central router needed!)
# On each robot, set up an IBSS ad-hoc wireless connection:
sudo ifconfig wlan0 down
sudo iwconfig wlan0 mode ad-hoc channel 6 essid "SWARMEDGE_MESH" ap 02:12:34:56:78:9A
sudo ifconfig wlan0 192.168.10.11 netmask 255.255.255.0 up

# 3. Launch AMR Node instances on each hardware unit:
# On Robot 1:
python3 swarm_edge_node.py --id R1 --x 0 --y 5 --rank 0

# On Robot 2:
python3 swarm_edge_node.py --id R2 --x 5 --y 0 --rank 1

# On Robot 3:
python3 swarm_edge_node.py --id R3 --x 10 --y 5 --rank 2
